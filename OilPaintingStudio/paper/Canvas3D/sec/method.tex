\section{Method}
\label{sec:method}

\paragraph{Design philosophy.}
A single painting-animation system must simultaneously satisfy four
\emph{conflicting} objectives: perceptual grouping of artistically rendered
content, geometric reconstruction of objects that have no real 3D extent,
user-specifiable composition, and temporally coherent stylized rendering.
Jointly optimizing a single network for all four is ill-posed because the
gradients of style fidelity and 3D structure pull in opposite directions.
Our central design principle is therefore \emph{modular separation of
concerns with explicit handshakes}: each objective is addressed by an
independently optimized specialist, and the four specialists communicate
only through well-defined, interpretable intermediate
representations---masks with depth-ordered layers, a metric-scaled
object-level 3D library, a composited 3D scene with disocclusion-filled
background, and finally a multi-view keyframe stream with paired
geometry and edge cues. This decomposition converts a conflicting joint
problem into four well-posed sub-problems, while the handshakes carry
exactly the information each downstream stage needs and no more,
preventing error compounding.

\paragraph{Problem statement.}
Given a painting scan $P \in \mathbb{R}^{H\times W \times 3}$ and a set of
hand-drawn line-art sketches $L = \{\ell_1, \dots, \ell_K\}$ specifying
target object positions, Canvas3D produces an animated video $V$ that
(i) preserves the artistic style of $P$, (ii) supports viewpoint
exploration of a 3D-composited scene, and (iii) places each foreground
object at its sketched location with geometrically grounded, temporally
coherent motion. The framework comprises four modules
(Figure~\ref{fig:overview}): \textbf{PaintParse} (perceptual grouping and
layer decomposition), \textbf{MonoSplat} (per-object 3D lifting and
library assembly), \textbf{SketchComp} (line-art-guided scene composition
with depth-aware inpainting), and \textbf{ArtiFlow} (geometry-grounded
video synthesis).

\paragraph{Notation.}
Let $\Omega = \{1,\dots,H\}\times\{1,\dots,W\}$ denote the pixel grid of
the painting. For the $i$-th foreground object
($i \in \{1,\dots,K\}$), $M_i \in \{0,1\}^{H\times W}$ is its binary
mask and $\Omega_i = \{p \in \Omega \mid M_i(p) = 1\}$ the associated
pixel set; the residual background mask is
$M_{\mathrm{bg}} \in \{0,1\}^{H\times W}$. The (metric-aligned)
monocular depth map is $D_{\mathrm{mono}} : \Omega \to \mathbb{R}_{+}$;
each object is summarized by its \emph{median} depth
$d_i = \operatorname{median}\{D_{\mathrm{mono}}(p) \mid p \in
\Omega_i\}$, used for metric recovery, together with a robust depth
range $[d_i^{\mathrm{near}}, d_i^{\mathrm{far}}]$ (the 10th and 90th
percentiles over $\Omega_i$), used only for layer ordering. The 3D
representation of object $i$ is $\mathcal{G}_i$ and the assembled
library is $\{\mathcal{G}_1,\dots,\mathcal{G}_K\}$; the composited
scene and its depth map are $\mathcal{S}^{*}$ and $\mathcal{D}^{*}$;
$\Omega_{\mathrm{gap}}$ is the disocclusion region; and
$\{I_t, D_t, E_t\}_{t=1}^{T}$ are the multi-view RGB, depth, and edge
keyframes rendered along camera poses $\{\mathbf{T}_t\}_{t=1}^{T}$. The
focal length in pixels is $f$. Throughout, $\pi(\cdot)$ denotes the
perspective projection onto the image plane and
$\operatorname{IoU}(\cdot,\cdot)$ the intersection-over-union between
two binary regions.

\subsection{PaintParse: Art-Domain Perceptual Grouping and Layer Ordering}
\label{sec:method:paintparse}

\paragraph{Design rationale.}
The downstream 3D and composition stages require two distinct outputs
that no single head of an off-the-shelf detector provides:
\emph{(i)} foreground masks whose boundaries respect artistic intent
rather than photographic edges, and \emph{(ii)} a depth-consistent
layer ordering that disambiguates which object lies in front of which.
PaintParse therefore couples a domain-adapted instance segmentation
backbone with a parallel monocular depth estimator and fuses their
outputs through a percentile-based layer assignment. The key design
choice is \emph{decoupling semantic grouping from depth ordering}: the
segmentation network is responsible only for ``what is an object,''
while the depth estimator is responsible only for ``how far away it
is,'' and the layer-assignment logic simply reads both off. This
avoids asking either network to solve a problem it was not trained
for.

\paragraph{Instance segmentation with art-domain adaptation.}
We adopt a one-stage instance segmentation backbone pre-trained on
natural-image data~\cite{jocher2023yolov8}. Direct transfer to
paintings yields poor mask quality because brushstroke textures
produce spurious edge responses and depth-ambiguous compositions
confound object--background separation. We mitigate this through a
two-stage adaptation protocol: a \emph{low-rank
adaptation}~\cite{hu2022lora} of the detection backbone, which injects
art-domain statistics while constraining the parameter drift that would
cause catastrophic forgetting, followed by \emph{full fine-tuning} of
the mask-prediction head to align boundary prediction with painterly
edges. Given input $P$, the adapted model outputs $K$ binary masks
$\{M_i\}_{i=1}^{K}$, ordered such that
$\Omega_i \cap \Omega_j = \varnothing$ for $i \neq j$ (overlapping
detections are suppressed by non-maximum suppression and mask
deduplication). The residual background mask is the complement of the
foreground union:
\begin{equation}
    M_{\mathrm{bg}}(p) \;=\; 1 \,-\, \max_{i\in\{1,\dots,K\}} M_i(p),
    \qquad \forall p \in \Omega.
    \label{eq:bg_mask}
\end{equation}

\paragraph{Monocular depth and layer assignment.}
In parallel, a monocular depth
estimator~\cite{yang2024depthanything} produces a relative depth map
$D_{\mathrm{mono}} : \Omega \to \mathbb{R}_{+}$ (nearness-coded: smaller
values indicate closer content). For each detected object $i$ we
summarize its depth by the 10th and 90th percentiles of
$\{D_{\mathrm{mono}}(p) \mid p \in \Omega_i\}$, yielding the robust
range $[d_i^{\mathrm{near}}, d_i^{\mathrm{far}}]$ that is insensitive
to outlier pixels at mask boundaries, and by the median
$d_i = \operatorname{median}\{D_{\mathrm{mono}}(p) \mid p \in
\Omega_i\}$, which the MonoSplat stage consumes for metric recovery.
The layer index of each object is determined by the depth percentile of
its mask centroid, i.e., object $i$ is assigned a \emph{lower} (front)
layer than object $j$ whenever the centroid depth of $i$ is smaller.
This percentile-based assignment is preferable to a per-pixel depth
sort because it treats each object as an atomic unit, preserving its
identity under the rigid transformations that subsequent composition
applies.

\subsection{MonoSplat: Per-Object 3D Lifting and Library Assembly}
\label{sec:method:monosplat}

\paragraph{Design rationale.}
A whole-scene 3D reconstruction would entangle object geometry with
background texture, making independent repositioning impossible. We
therefore lift each foreground object \emph{separately} into a 3D
representation and assemble a \emph{reusable library}, indexed by
semantic label and original painting coordinates. Two design choices
follow from this principle. First, we use a \emph{feed-forward
single-image 3D predictor} rather than per-scene optimization, because
per-object optimization would be prohibitively expensive and would
overfit to a single viewpoint. Second, because feed-forward predictors
operate in a normalized canonical space without metric scale, we
\emph{recover metric dimensions from the PaintParse depth cue}, so that
objects from different paintings can later be composed in a common
metric frame.

\paragraph{Feed-forward lifting.}
For each foreground object $i$, we crop the painting to its mask,
$P_i = P \odot M_i$, and feed the crop into a feed-forward single-image
3D predictor~\cite{tripo2025tritosplat} that regresses a set of $N_i$
anisotropic 3D Gaussians
directly from one image. Each Gaussian $j$ is parameterized by a center
$\boldsymbol{\mu}_j \in \mathbb{R}^3$, scales
$\mathbf{s}_j \in \mathbb{R}_{+}^{3}$, rotation unit quaternion
$\mathbf{q}_j \in \mathbb{S}^{3} \subset \mathbb{R}^{4}$, opacity
$\alpha_j \in [0,1]$, and view-dependent color coefficients, and the
object is rendered by $\alpha$-composited Gaussian splatting. The
predictor's generative prior, trained on diverse 3D assets, generalizes
to the missing side and back of objects that are not visible in the
painting---a critical capability for the rotations that downstream
composition imposes.

\paragraph{Metric rescaling.}
The predictor outputs geometry in a normalized canonical space, i.e.,
it estimates shape $\widehat{\mathcal{G}}_i$ up to an unknown similarity
transform. To place objects from different paintings (and different
object classes) in a single metric frame, we recover the scale factor
from the pinhole imaging relation. Under the standard thin-lens
approximation, the physical height $h_i$ of an object, its image-plane
extent $h_i^{\mathrm{px}}$ in pixels, its depth $d_i$, and the focal
length $f$ (in pixels) satisfy
$h_i = d_i \, h_i^{\mathrm{px}} / f$. We therefore rescale the
canonical geometry by
\begin{equation}
    \lambda_i \;=\; \frac{d_i \cdot h_i^{\mathrm{px}}}{f \cdot
    h_i^{\mathrm{can}}},
    \label{eq:metric_scale}
\end{equation}
where $d_i$ is the median depth of object $i$ (from PaintParse),
$h_i^{\mathrm{px}}$ is its mask height in the painting, and
$h_i^{\mathrm{can}}$ is the height of its bounding volume in the
predictor's canonical output space; the ratio
$h_i^{\mathrm{px}}/h_i^{\mathrm{can}}$ converts the canonical unit to
pixels before the depth relation maps it to metric units. Each Gaussian
center and scale is rescaled as
$\boldsymbol{\mu}_j \mapsto \lambda_i \boldsymbol{\mu}_j$,
$\mathbf{s}_j \mapsto \lambda_i \mathbf{s}_j$, producing a
metric-consistent 3D object $\mathcal{G}_i =
\lambda_i \widehat{\mathcal{G}}_i$. The resulting library
$\{\mathcal{G}_1, \dots, \mathcal{G}_K\}$ is keyed by semantic label
and original painting coordinates, enabling selective retrieval and
cross-painting composition in future work.

\subsection{SketchComp: Line-Art-Guided 3D Scene Composition}
\label{sec:method:sketchcomp}

\paragraph{Design rationale.}
The user's compositional intent---which object goes where---is most
naturally expressed as a sketch, but existing diffusion-based sketch
control treats sketches as appearance or boundary cues rather than
spatial constraints. SketchComp inverts this convention: a sketch
\emph{is} the target, and a 3D object is \emph{geometrically
registered} to it. Two sub-problems must then be solved:
\emph{(i)} finding the rigid transformation that aligns each 3D
object's rendered silhouette with its sketched contour, and
\emph{(ii)} filling the disocclusion gaps that open up when an object
moves away from its original position, while preserving the
background's artistic texture. We solve the first with a
silhouette-overlap objective that combines region IoU with boundary
Chamfer distance, and the second with a depth-conditioned inpainting
module whose geometric conditioning prevents the texture-smear
artifact that naive inpainting produces on paintings.

\paragraph{Contour-based rigid registration.}
Each sketch $\ell_i$ is converted to a binary edge map via Canny
detection, and its external contour $\Gamma_i \subset \Omega$ is
extracted. For each 3D object $\mathcal{G}_i$, we search for the rigid
transformation---in-plane rotation $\theta \in [0,2\pi)$, 2D translation
$\mathbf{t} \in \mathbb{R}^2$, and uniform scale $s \in
\mathbb{R}_{+}$---that best aligns the rendered silhouette with
$\Gamma_i$:
\begin{align}
    (\theta_i^{*},\, \mathbf{t}_i^{*},\, s_i^{*}) &=
      \arg\min_{\theta,\,\mathbf{t},\,s}\;
      \mathcal{L}_{\mathrm{cont}}(\theta,\mathbf{t},s),
      \label{eq:contour_reg}\\
    \mathcal{L}_{\mathrm{cont}}(\theta,\mathbf{t},s) &=
      \underbrace{1 - \operatorname{IoU}\bigl(R_i(\theta,\mathbf{t},s),
        \,R(\Gamma_i)\bigr)}_{\text{region}}
      \nonumber\\[-2pt]
    &\quad + \lambda_c \,
      \underbrace{\mathrm{CD}\bigl(\partial R_i(\theta,\mathbf{t},s),
        \,\Gamma_i\bigr)}_{\text{boundary}},
      \nonumber
\end{align}
where $R_i(\theta,\mathbf{t},s)$ is the binary silhouette obtained by
rendering $\mathcal{G}_i$ from the canonical camera under the candidate
transform, $R(\Gamma_i)$ is the filled region enclosed by the sketched
contour, $\partial R_i$ denotes its boundary curve, $\mathrm{CD}(A,B) =
\tfrac{1}{|A|}\sum_{a \in A} \min_{b \in B} \|a - b\|_2$ is the directed
Chamfer distance (in pixels), and $\lambda_c > 0$ balances the two
terms. The IoU term rewards global region agreement, while the Chamfer
term penalizes boundary misalignment; their combination avoids the
degenerate solutions that either term alone admits (a small silhouette
placed inside the contour attains zero Chamfer error at the wrong
scale, while region-only matching tolerates topologically incorrect
regions whose interiors overlap). Optimization proceeds by gradient
descent with iterative re-rendering, so that each step's gradient is
computed against the current geometry rather than a stale
approximation; the resulting transform is applied to $\mathcal{G}_i$
to obtain the placed object $\mathcal{G}'_i$.

\paragraph{Scene assembly and disocclusion inpainting.}
Each registered object is placed at its PaintParse-derived depth $d_i$,
and objects are rendered in depth-sorted (painter's) order against a
background layer reconstructed from the RGB-D map. Object displacement
creates \emph{disocclusion gaps}: background pixels that become newly
visible because the surface content previously occluded by the object
in its original position is not recorded in the single input image. We
detect the gap as the set of pixels inside the original foreground
support whose projected scene content is missing:
\begin{equation}
    \Omega_{\mathrm{gap}} \;=\;
    \Bigl\{\,p \in \Omega \;\Big|\;
        \mathcal{V}_{\mathcal{S}}(p) = 0 \Bigr\},
    \label{eq:gap}
\end{equation}
where $\mathcal{V}_{\mathcal{S}}(p) = 1$ if and only if at least one 3D
surface in the assembled scene $\mathcal{S}$ projects onto pixel $p$
with valid appearance (in practice, pixels covered by the original
$M_{\mathrm{bg}}$ support or by any placed object). Filling
$\Omega_{\mathrm{gap}}$ while preserving artistic texture coherence is
critical. We employ a depth-conditioned inpainting diffusion
pipeline~\cite{lugmayr2022repaint,rombach2022ldm}: a masked composite
(with $\Omega_{\mathrm{gap}}$ zeroed and dilated to expose surrounding
context as texture reference) is fed into a latent inpainting diffusion
model, with a depth-conditioned control
branch~\cite{zhang2023controlnet} conditioned on the composited depth
map $\mathcal{D}$. The depth
conditioning is the key design choice: it constrains the inpainted
content to follow the background's geometric structure, preventing the
texture-smear artifact that arises when an unconstrained inpainter
fills a painting gap with semantically plausible but geometrically
inconsistent content. The resulting scene $\mathcal{S}^{*}$ and its
depth map $\mathcal{D}^{*}$ are forwarded to ArtiFlow.

\subsection{ArtiFlow: Geometry-Grounded Video Synthesis}
\label{sec:method:artiflow}

\paragraph{Design rationale.}
Pure image-to-video diffusion applied to a painting produces
plausible per-frame motion but suffers two failure modes: geometric
drift (objects wander off their intended trajectories) and boundary
melting (brushstroke edges blur across frames). ArtiFlow addresses
both by \emph{grounding the diffusion in explicit 3D geometry}: the
composited scene is rendered into a multi-view keyframe stream that
provides three control signals---RGB appearance, depth geometry, and
edge boundary---which are injected into a video diffusion backbone as
dual control branches. The depth branch anchors each frame to the 3D
structure of $\mathcal{S}^*$, suppressing geometric drift; the edge
branch preserves brushstroke-level boundary detail, suppressing
boundary melting. The combined conditioning converts the diffusion
backbone from a free-form motion generator into a \emph{geometry-anchored
refiner} whose temporal freedom is restricted to the residual motion
that the 3D rendering does not already specify.

\paragraph{Multi-view keyframe rendering.}
Given $\mathcal{S}^{*}$, we define a smooth camera trajectory through
$T$ poses $\{\mathbf{T}_t\}_{t=1}^{T} \subset SE(3)$, generated by
cubic B-spline interpolation between a canonical front-facing start
pose and a user-specified or auto-generated end pose. At each step $t$,
differentiable
rasterization~\cite{kerbl20233dgs,huang20242dgs} produces an RGB frame
$I_t = \mathcal{R}_{\mathrm{rgb}}(\mathcal{S}^{*},\mathbf{T}_t)$, a
depth map $D_t = \mathcal{R}_{\mathrm{depth}}
(\mathcal{S}^{*},\mathbf{T}_t)$, and an edge map
$E_t = \operatorname{Edge}(I_t)$, where edges are extracted with
HED~\cite{xie2015hed}. The keyframe count is set to
$T = 8N+1$ for some $N \in \mathbb{N}$, aligning the keyframe stream
with the temporal chunking of modern video diffusion backbones.

\paragraph{Dual-conditioned video diffusion.}
Two parallel control branches inject geometric and boundary cues into
the video diffusion backbone~\cite{ho2022video,blattmann2023svd},
following the ControlNet
design~\cite{zhang2023controlnet,wang2025easycontrol}. The depth branch, conditioned on
$\{D_t\}$ at strength $s_d \in [0,1]$, anchors each frame to the 3D
structure of $\mathcal{S}^{*}$, preventing the geometric drift that
plagues pure I2V models. The edge branch, conditioned on $\{E_t\}$ at
strength $s_e \in [0,1]$, preserves brushstroke-level boundary detail,
suppressing the boundary-melting artifact common when diffusion is
applied to paintings. Writing $\mathbf{c}_t^{\mathrm{depth}}$ and
$\mathbf{c}_t^{\mathrm{edge}}$ for the control features produced by the
two branches at frame $t$, the combined conditioning signal is
\begin{equation}
    \mathbf{c}_t \;=\; s_d \, \mathbf{c}_t^{\mathrm{depth}}
                   \;+\; s_e \, \mathbf{c}_t^{\mathrm{edge}},
    \qquad t = 1, \dots, T,
    \label{eq:dual_control}
\end{equation}
injected at the corresponding residual connections of the backbone.
The two feature streams share a common embedding space (both control
encoders project into the backbone's residual channels), so the
weighted sum in Eq.~\ref{eq:dual_control} is well defined.
Temporal coherence is further reinforced by \emph{first-frame latent
anchoring}: the initial latent of frame $t>1$ is set to the encoded
first frame plus a position-encoded Gaussian perturbation
$\boldsymbol{\epsilon}_t \sim \mathcal{N}(0, \sigma_t^2 \mathbf{I})$
with $\sigma_t = \sigma_{\max}\, (t-1)/(T-1)$ growing linearly in the
temporal distance $|t-1|$, so that motion freedom increases smoothly
away from the anchored first frame while the drift budget remains
bounded by $\sigma_{\max}$.

\paragraph{Per-object motion verification.}
After generation, we verify motion fidelity by computing the IoU
between rendered object masks and segmented masks in each generated
frame. If the average IoU falls below a threshold $\tau_\text{mo}$,
the pipeline triggers a \emph{re-composition} step (dashed arrow in
Figure~\ref{fig:overview}) that adjusts control strengths or
regenerates the affected segment with stronger depth conditioning,
closing the loop between composition and synthesis.

\subsection{Pipeline Algorithm and Implementation}
\label{sec:method:algo}

Algorithm~\ref{alg:canvas3d} summarizes the full Canvas3D pipeline. The
modular design enables \emph{fault-tolerant resumption}: each stage
saves its outputs as checkpoint files, so any specialist can be
upgraded or replaced without re-running downstream stages---for
example, swapping in a stronger 3D predictor or a different video
backbone. Peak memory consumption across all four stages is kept
within a single 24\,GB budget through serial model loading with
explicit memory release between stages, which is feasible precisely
because the handshake representations between stages are small,
serializable tensors rather than shared activations.

\begin{algorithm}[t]
\caption{Canvas3D Pipeline}
\label{alg:canvas3d}
{\footnotesize
\begin{algorithmic}[1]
\REQUIRE Painting $P$; line-art sketches $L=\{\ell_i\}_{i=1}^{K}$;
         thresholds $\lambda_c$, $s_d$, $s_e$, $\tau_\text{mo}$
\ENSURE Animated video $V$ with per-object motion
\medskip
\STATE \textbf{// Stage 1 --- PaintParse: perceptual grouping + layers}
\STATE $\{M_i\}_{i=1}^{K} \gets \text{SegAdapt}(P)$
    \hfill $\triangleright$ art-domain-adapted segmentation
\STATE $M_\text{bg} \gets 1 - \max_i M_i$
    \hfill $\triangleright$ Eq.~\ref{eq:bg_mask}
\STATE $D_\text{mono} \gets \text{DepthEstimator}(P)$
\STATE \textbf{for} $i = 1$ \textbf{to} $K$ \textbf{do}
\STATE \quad $\Omega_i \gets \{p \mid M_i(p){=}1\}$;
    $d_i \gets \text{Median}\{D_\text{mono}(p) \mid p \in \Omega_i\}$
\STATE \textbf{end for}
\medskip
\STATE \textbf{// Stage 2 --- MonoSplat: per-object 3D lifting}
\STATE \textbf{for} $i = 1$ \textbf{to} $K$ \textbf{do}
\STATE \quad $\widetilde{\mathcal{G}}_i \gets
    \text{FeedForward3D}(P \odot M_i)$
    \hfill $\triangleright$ canonical-space 3D
\STATE \quad $\lambda_i \gets d_i \cdot h_i^\text{px}/f$
    \hfill $\triangleright$ Eq.~\ref{eq:metric_scale}
\STATE \quad $\mathcal{G}_i \gets \lambda_i \cdot \widetilde{\mathcal{G}}_i$
    \hfill $\triangleright$ metric rescaling
\STATE \quad $\text{Lib}[i] \gets (\mathcal{G}_i,\, d_i,\, M_i)$
\STATE \textbf{end for}
\medskip
\STATE \textbf{// Stage 3 --- SketchComp: composition + inpainting}
\STATE \textbf{for} $i = 1$ \textbf{to} $K$ \textbf{do}
\STATE \quad $\Gamma_i \gets \text{ExternalContour}(\text{Canny}(\ell_i))$
\STATE \quad $(\theta_i^*,\mathbf{t}_i^*,s_i^*) \gets
    \arg\min \mathcal{L}_\text{cont}(\mathcal{G}_i,\Gamma_i)$
    \hfill $\triangleright$ Eq.~\ref{eq:contour_reg}
\STATE \quad $\mathcal{G}'_i \gets
    \text{Transform}(\mathcal{G}_i,\theta_i^*,\mathbf{t}_i^*,s_i^*)$
\STATE \textbf{end for}
\STATE $\mathcal{S},\mathcal{D} \gets
    \text{DepthSort}(\{\mathcal{G}'_i\},\,M_\text{bg},\,\{d_i\})$
\STATE $\Omega_\text{gap} \gets
    \text{DetectGap}(\mathcal{S},\,M_\text{bg}^\text{orig})$
    \hfill $\triangleright$ Eq.~\ref{eq:gap}
\STATE $\mathcal{S}^*,\mathcal{D}^* \gets
    \text{DepthInpaint}(\mathcal{S},\,\mathcal{D},\,\Omega_\text{gap})$
\medskip
\STATE \textbf{// Stage 4 --- ArtiFlow: geometry-grounded video synthesis}
\STATE $\{\mathbf{T}_t\}_{t=1}^{T} \gets \text{BSplineTrajectory}(T)$
\STATE \textbf{for} $t = 1$ \textbf{to} $T$ \textbf{do}
\STATE \quad $I_t \gets \mathcal{R}_\text{rgb}(\mathcal{S}^*,\mathbf{T}_t)$;
\STATE \quad $D_t \gets \mathcal{R}_\text{depth}(\mathcal{S}^*,\mathbf{T}_t)$;
\STATE \quad $E_t \gets \text{Edge}(I_t)$
\STATE \textbf{end for}
\STATE $V \gets \text{VideoDiffusion}\bigl(\{I_t\},\,s_d\!\cdot\!D_t
    + s_e\!\cdot\!E_t\bigr)$
    \hfill $\triangleright$ Eq.~\ref{eq:dual_control}
\STATE \textbf{if} $\text{MotionIoU}(V) < \tau_\text{mo}$ \textbf{then}
    \quad \textbf{go to} Stage 3
    \hfill $\triangleright$ re-composition loop
\STATE \textbf{return} $V$
\end{algorithmic}}
\end{algorithm}
